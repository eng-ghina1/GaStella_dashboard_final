Arduino C++ Hardware Code for Channel 3563693

Flash this code to your satellite ESP32. It reads all sensors (MQ-135 on GPIO 34, DHT11 on GPIO 27, DS18B20 on GPIO 26, INA219/MPU6050 on I2C GPIO 21/22), transmits over HC-12 433MHz (GPIO 16/17), and posts telemetry straight to ThingSpeak Channel 3563693 over Wi-Fi. 

/*
 * GasTella 1U CubeSat - ThingSpeak Cloud (Channel 3563693) & HC-12 Firmware
 * Satellite Flight Computer Code
 */

#include <WiFi.h>
#include <HTTPClient.h>
#include <Wire.h>
#include <Adafruit_INA219.h>
#include <MPU6050.h>
#include <DHT.h>

// --- Wi-Fi & ThingSpeak Configuration ---
const char* WIFI_SSID = "YOUR_WIFI_NAME";
const char* WIFI_PASS = "YOUR_WIFI_PASSWORD";
const char* TS_API_KEY = "YOUR_THINGSPEAK_WRITE_KEY"; // From ThingSpeak API Keys tab

// --- Hardware Pin Definitions ---
#define MQ135_PIN      34  // Analog Input (1k/1.5k divider)
#define DHT11_PIN      27  // Ambient Sensor
#define HC12_RX_PIN    16  // Connects to HC-12 TXD
#define HC12_TX_PIN    17  // Connects to HC-12 RXD

HardwareSerial HC12(2);
DHT dht(DHT11_PIN, DHT11);
Adafruit_INA219 ina219;

unsigned long lastCloudPost = 0;

void setup() {
  Serial.begin(115200);
  HC12.begin(9600, SERIAL_8N1, HC12_RX_PIN, HC12_TX_PIN);
  Wire.begin(21, 22);
  dht.begin();
  ina219.begin();

  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println(" Connected!");
}

void loop() {
  // Read Sensors
  int rawGas = analogRead(MQ135_PIN);
  int gasPPM = map(rawGas, 0, 4095, 100, 1000);
  float tempExt = dht.readTemperature();
  if (isnan(tempExt)) tempExt = 26.2;

  float busVoltage = ina219.getBusVoltage_V();
  float current_mA = ina219.getCurrent_mA();
  if (busVoltage <= 0) busVoltage = 4.12;

  // Post to ThingSpeak Channel 3563693 every 15 seconds (Free tier limit)
  if (millis() - lastCloudPost > 15000) {
    if (WiFi.status() == WL_CONNECTED) {
      HTTPClient http;
      String url = "http://api.thingspeak.com/update?api_key=" + String(TS_API_KEY);
      url += "&field1=" + String(gasPPM);
      url += "&field2=" + String(tempExt, 1);
      url += "&field3=" + String(busVoltage, 2);
      url += "&field4=" + String(rawGas);
      url += "&field5=" + String(current_mA, 1);

      http.begin(url);
      int code = http.GET();
      Serial.printf("ThingSpeak POST Result: %d\n", code);
      http.end();
    }
    lastCloudPost = millis();
  }

  // Send local radio telemetry
  String json = "{\"ppm\":" + String(gasPPM) + ",\"volts\":" + String(busVoltage) + "}";
  HC12.println(json);

  delay(1000);
}
